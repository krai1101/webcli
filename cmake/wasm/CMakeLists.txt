# A browser may grow exported memory from JavaScript even when the module has
# no memory.grow opcode. WAMR's standalone optimization collapses those pages
# and their maximum. Disable that optimization in our generated loader copy.
set(WAMR_LOADER "${WAMR_ROOT_DIR}/core/iwasm/interpreter/wasm_loader.c")
file(READ "${WAMR_LOADER}" WAMR_LOADER_TEXT)
string(FIND "${WAMR_LOADER_TEXT}" "if (!module->possible_memory_grow) {" WAMR_GROW_CHECK)
if(WAMR_GROW_CHECK LESS 0)
  message(FATAL_ERROR "Pinned WAMR loader no longer matches the host-memory adapter")
endif()
string(REPLACE "if (!module->possible_memory_grow) {"
  "if (false) { /* WebCLI: host JavaScript can grow exported memory. */"
  WAMR_LOADER_TEXT "${WAMR_LOADER_TEXT}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/wasm_loader.c"
  CONTENT "${WAMR_LOADER_TEXT}" @ONLY)
list(REMOVE_ITEM WAMR_RUNTIME_LIB_SOURCE "${WAMR_LOADER}")
list(APPEND WAMR_RUNTIME_LIB_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/wasm_loader.c")
add_library(webcli_wasm STATIC ${WAMR_RUNTIME_LIB_SOURCE}
  "${PROJECT_SOURCE_DIR}/src/runtime/wasm/wamr_adapter.c")
target_include_directories(webcli_wasm PUBLIC "${WAMR_ROOT_DIR}/core/iwasm/include"
  "${PROJECT_SOURCE_DIR}/src/runtime/wasm")
target_include_directories(webcli_wasm PRIVATE "${WAMR_ROOT_DIR}/core/iwasm/interpreter")
if(MSVC)

endif()

endif()
